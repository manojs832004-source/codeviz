def selection_sort(arr):
    n = len(arr)
    for i in range(n):
        min_index = i                        # Assume current position holds minimum
        for j in range(i + 1, n):
            if arr[j] < arr[min_index]:   # Found a smaller element?
                min_index = j                # Update min_index
        arr[i], arr[min_index] = arr[min_index], arr[i]  # Swap

arr = [29, 10, 14, 37, 13]
selection_sort(arr)
print(arr)   # → [10, 13, 14, 29, 37]
